# DeepShield AI — runtime dependencies
#
#   Windows:      venv\Scripts\pip install -r requirements.txt
#   macOS/Linux:  venv/bin/pip install -r requirements.txt
#
# This is the lean install (~160 MB). The model runs as ONNX through
# OpenCV's DNN module, so PyTorch is NOT needed to use the app.

flask>=3.1
opencv-python-headless>=4.8   # face detection, video frames, and the model
                              # itself. HEADLESS on purpose: the app calls no
                              # cv2 GUI function, and the desktop build needs
                              # libGL, which a server container does not have.
                              # Installing the wrong one is the single most
                              # common way a deploy of this shape fails.
                              # Already have opencv-python? Uninstall it first:
                              # the two packages conflict.
pillow>=10.0
numpy>=1.24


# ---------------------------------------------------------------------
# Optional extras — install only if you need them
# ---------------------------------------------------------------------
#
# 1. Training, or exporting a new ONNX from a .pth (scripts/export_onnx.py):
#      pip install torch torchvision onnx onnxscript
#    On Linux add --index-url https://download.pytorch.org/whl/cpu,
#    otherwise pip pulls the ~2 GB CUDA build for nothing.
#    The backend also falls back to a .pth checkpoint if torch is present
#    and no ONNX export exists.
#
# 2. Ensemble verifiers (two pretrained HuggingFace models, ~1 GB, opt-in
#    with DS_VERIFIERS=1):
#      pip install transformers torch
#
# 3. Serving it in production:
#      pip install gunicorn
#    Flask's built-in server is single-threaded and not meant to face the
#    internet. See docs/DEPLOY.md for the exact command and why it uses
#    one worker.
#
# 4. Collecting feedback and analytics in a database (optional):
#      pip install "psycopg[binary]"
#    Then set DATABASE_URL. Without both, feedback appends to a JSONL file
#    exactly as before and nothing else is recorded - local development
#    needs no database. See docs/ANALYTICS.md for what is and is not stored.
#
# 5. Running the test suite:
#      pip install pytest
#    Then `python -m pytest` from the project root. The suite needs no
#    server and no network; it builds its own test media from the sample
#    faces already in training/.
#
# 6. Server-side PDF reports (not used — the report page prints from the
#    browser):
#      pip install reportlab
